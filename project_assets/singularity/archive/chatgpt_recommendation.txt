Bootstrap: docker
From: ubuntu:22.04

%post
    # Update and install base dependencies
    apt-get update && apt-get install -y \
        build-essential \
        cmake \
        git \
        python3-dev \
        python3-pip \
        python3-venv \
        libpython3.10 \
        libpython3.10-dev \
        libglu1-mesa \
        x11-xserver-utils \
        ros-humble-desktop \
        python3-colcon-common-extensions \
        && rm -rf /var/lib/apt/lists/*


    # Install ROS 2 Humble dependencies
    apt-get update && apt-get install -y \
        locales \
        wget \
        curl \
        gnupg2 \
        lsb-release \
        && rm -rf /var/lib/apt/lists/*

    # Set locale
    locale-gen en_US en_US.UTF-8
    update-locale LC_ALL=en_US.UTF-8 LANG=en_US.UTF-8
    export LANG=en_US.UTF-8

    # Add ROS 2 GPG key and repository
    curl -sSL https://raw.githubusercontent.com/ros/rosdistro/master/ros.key | apt-key add -
    echo "deb [arch=amd64] http://packages.ros.org/ros2/ubuntu $(lsb_release -cs) main" > /etc/apt/sources.list.d/ros2-latest.list

    # Install ROS 2 Humble
    apt-get update && apt-get install -y \
        ros-humble-desktop \
        python3-colcon-common-extensions \
        && rm -rf /var/lib/apt/lists/*

    # Source ROS 2 setup script
    echo "source /opt/ros/humble/setup.bash" >> ~/.bashrc

    # Upgrade pip and install Python packages
    pip3 install --upgrade pip
    pip3 install casadi

    # Clone and install acados
    cd /root
    git clone https://github.com/acados/acados.git
    cd acados
    git submodule update --init --recursive
    mkdir build
    cd build
    cmake -DACADOS_WITH_QPOASES=ON -DACADOS_WITH_OSQP=ON ..
    make -j$(nproc)
    make install

    # Set environment variables for acados
    echo 'export ACADOS_SOURCE_DIR=/root/acados' >> /root/.bashrc
    echo 'export LD_LIBRARY_PATH=/root/acados/lib:$LD_LIBRARY_PATH' >> /root/.bashrc
    pip3 install -e /root/acados/interfaces/acados_template

    # Install gym-pybullet-drones
    pip3 install git+https://github.com/utiasDSL/gym-pybullet-drones.git

%environment
    # Source ROS and set environment variables
    source /opt/ros/humble/setup.bash
    export ACADOS_SOURCE_DIR=/root/acados
    export LD_LIBRARY_PATH=/root/acados/lib:$LD_LIBRARY_PATH

%runscript
    # Source ROS and launch the container
    source /opt/ros/humble/setup.bash
    exec "$@"

%test
    # Verify installation
    python3 -c "import casadi; import acados_template; import gym; print('Setup successful!')"
